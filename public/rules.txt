
rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ==========================================
    // ADMIN CHECK
    // Firebase Auth + admins/{uid}
    // ==========================================

    function isAdmin() {
      return request.auth != null
        && exists(
          /databases/$(database)/documents/admins/$(request.auth.uid)
        )
        && get(
          /databases/$(database)/documents/admins/$(request.auth.uid)
        ).data.role == "admin";
    }


    // ==========================================
    // ADMINS
    // ==========================================

    match /admins/{adminId} {
      allow read: if isAdmin();
      allow write: if isAdmin();
    }


    // ==========================================
    // TEACHERS
    // ==========================================

    match /teachers/{teacherId} {
      allow read: if true;
      allow create: if request.auth != null;
      allow update: if isAdmin() || (request.auth != null && request.auth.uid == teacherId);
      allow delete: if isAdmin();
    }


    // ==========================================
    // STUDENTS
    // ==========================================

    match /students/{studentId} {
      allow read: if true;
      allow write: if true;
    }


    // ==========================================
    // QUIZ METADATA
    // ==========================================

    match /quiz_metadata/{quizId} {
      allow read: if true;
      allow write: if true;
    }


    // ==========================================
    // QUIZ QUESTIONS
    // ==========================================

    match /quiz_questions/{quizId} {
      allow read: if true;
      allow write: if true;
    }


    // ==========================================
    // RESULTS
    // ==========================================

    match /results/{resultId} {
      allow create: if true;
      allow read: if true;
      allow update: if true;
      allow delete: if isAdmin();
    }


    // ==========================================
    // GAME ANSWERS (Student Submissions)
    // ==========================================

    match /game_answers/{answerId} {
      // Allows students to submit/log their answers and queries during live play
      allow read, write: if true;
    }


    // ==========================================
    // GAME SESSIONS (Lobbies & Realtime Sync)
    // ==========================================

    match /game_sessions/{sessionId} {
      // Allows teachers to create/update game state and students to sync in real-time
      allow read, write: if true;

      // Allow public read/write to ALL nested subcollections recursively
      match /{document=**} {
        allow read, write: if true;
      }
    }


    // ==========================================
    // COLLECTION GROUP QUERY PERMISSIONS
    // ==========================================

    // Explicitly enables collectionGroup queries for 'scores' across all sessions
    match /{path=**}/scores/{scoreId} {
      allow read, write: if true;
    }


    // ==========================================
    // SETTINGS
    // ==========================================
    
    match /settings/{settingId} {
      allow read: if true;
      allow write: if request.auth != null;
    }


    // ==========================================
    // DEFAULT DENY
    // ==========================================

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
